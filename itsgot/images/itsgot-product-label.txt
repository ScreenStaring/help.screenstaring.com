Code for Nutrtion Facts app by ScreenStaring.
For support email us at help@screenstaring.com

Add this file to your theme's snippets directory as product-label.liquid
You must also add the itsgot-product-label-embed-code.liquid file to your theme's snippets directory.

This file will render a product's label. If the product has variants then it will show the labels
for the selected variant. If the variant changes it will show its label, if any.

You can forgo this functionality and display a seperate dropdown using the display_variant_dropdown: option.
Or use a dropdown that you've created via the use_variant_dropdown: option and providing a CSS selctor that can be
use to retrieve it.

See below examples.

Usage:

  {% render "itsgot-product-label", product: product %}

Show a dropdown that allows the variant to be selected:

  {% render "itsgot-product-label", product: product, display_variant_dropdown: true %}

The dropdown will be rendered with a DOM ID of itsgot-variant-select

  {% render "itsgot-product-label", product: product, use_variant_dropdown: "#my_dropdown" %}

  <script>
   window.ItsGot = window.ItsGot || {};
   ItsGot.lastVariant = null;
   ItsGot.initialVariant = {{ product.selected_or_first_available_variant.id | json }};
   ItsGot.__variantSelected = function(id) {
     const labels = document.querySelectorAll('[data-itsgot-external-variant-id]');
     labels.forEach(e => e.classList.remove('itsgot-label-visible'));

     const selected = document.querySelector(`[data-itsgot-external-variant-id="${id}"]`);
     if(selected) selected.classList.add('itsgot-label-visible');
   };
  </script>

  {% elsif use_variant_dropdown != blank %}
    <script>
     (function() {
         function error(message) {
             console.error(`Nutrition Facts App: ${message}`);
         }

         try {
             const variantSelect = document.querySelector({{ use_variant_dropdown | json }});
             if(!variantSelect) {
                 error('cannot find a dropdown with the given CSS selector: {{ use_variant_dropdown | json }}');
             }
             else {
                 {% comment %}We use addEventListener as there may be other handlers{% endcomment %}
                 variantSelect.addEventListener('change', function(e) {
                     if(this.selectedIndex === -1) return;

                     const id = this.options[this.selectedIndex].value;
                     if(!id.match(/^\d+$/)) error(`option value must be the variant id, found: ${id}`);

                     ItsGot.__variantSelected(id);
                 });
             }
         }
         catch(e) {
             error(e.message);
         }
     })();
    </script>
  {% else %}
    {% comment %}Can't rely on location.href because this may be called before it changes so we do this on every change{% endcomment %}
    <script>
     (function() {
       new MutationObserver(_ => {
         const id = new URL(location.href).searchParams.get('variant');
         if(id && id !== ItsGot.lastVariant) {
           ItsGot.__variantSelected(id);
           ItsGot.lastVariant = id;
         }
       }).observe(document, {subtree: true, childList: true});
     })();
    </script>
  {% endif %}
